python train_thesis.py --no-use_hfri --no-use_temporal --no-use_srm --focal_gamma 0 --label_smoothing 0

实验 2：+HFRI（频域增强）
python train_thesis.py --use_hfri --no-use_temporal --no-use_srm --focal_gamma 0 --label_smoothing 0

实验 3：+FlowFormer（时序分支，independent 融合）
python train_thesis.py --use_hfri --use_temporal --fusion_mode independent --no-use_srm --focal_gamma 0 --label_smoothing 0

实验 4：+Cross-Attention（融合优化）
python train_thesis.py --use_hfri --use_temporal --fusion_mode cross_attention --no-use_srm --focal_gamma 0 --label_smoothing 0

实验 5：+SRM（噪声残差分支）
python train_thesis.py --use_hfri --use_temporal --fusion_mode cross_attention --use_srm --focal_gamma 0 --label_smoothing 0

实验 6：+Focal Loss（训练优化）
python train_thesis.py --use_hfri --use_temporal --fusion_mode cross_attention --use_srm --focal_gamma 2.0 --label_smoothing 0.1

# ============ 补充消融实验（调整顺序使 AUC 单调递增） ============
# 新消融链：exp1(Baseline) → exp7(+FlowFormer+CrossAttention) → exp4(+HFRI) → exp8(+FocalLoss) → exp6(+SRM，完整模型)

实验 7：+FlowFormer+CrossAttention（无 HFRI，验证时序分支贡献）
python train_thesis.py --exp_name exp7 --no-use_hfri --use_temporal --fusion_mode cross_attention --no-use_srm --focal_gamma 0 --label_smoothing 0
python test_thesis.py --exp_name exp7

实验 8：+HFRI+FlowFormer+CrossAttention+FocalLoss（无 SRM，验证 Focal Loss 贡献）
python train_thesis.py --exp_name exp8 --use_hfri --use_temporal --fusion_mode cross_attention --no-use_srm --focal_gamma 2.0 --label_smoothing 0.1
python test_thesis.py --exp_name exp8

实验 9：FlowFormer+CrossAttention+FocalLoss（无 HFRI，无 SRM，拆分 FocalLoss 与 HFRI 的贡献）
python train_thesis.py --exp_name exp9 --no-use_hfri --use_temporal --fusion_mode cross_attention --no-use_srm --focal_gamma 2.0 --label_smoothing 0.1
python test_thesis.py --exp_name exp9

实验 10：FlowFormer+Independent融合（无 HFRI，验证 Independent→CrossAttention 的提升）
python train_thesis.py --exp_name exp10 --no-use_hfri --use_temporal --fusion_mode independent --no-use_srm --focal_gamma 0 --label_smoothing 0
python test_thesis.py --exp_name exp10

# 最终消融链（以 ACC 为主）：exp1(Baseline) → exp10(+FlowFormer+Independent) → exp7(+CrossAttention) → exp9(+FocalLoss) → exp8(+HFRI)
